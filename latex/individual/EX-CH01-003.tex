\begin{exercice}{EX-CH01-003}{Décomposition en facteurs premiers}

\begin{mdframed}[backgroundcolor=vlmgray, innerleftmargin=10pt, innerrightmargin=10pt, innertopmargin=6pt, innerbottommargin=6pt, skipabove=-5pt, skipbelow=5pt]
\textbf{Théorème fondamental de l'arithmétique :}

Chaque entier supérieur à 1 est premier ou un produit de nombres premiers ; la factorisation est unique, mis à part l'ordre des facteurs.
\end{mdframed}


\medskip
Décompose chaque nombre ci-dessous en un produit de facteurs premiers.

\medskip
\def\arraystretch{1.7}
\begin{tabularx}{\linewidth}{*{3}{X}}
\textbf{A.} & \textbf{B.} & \textbf{C.} \\
$255  =$ \sol{\dotfill}{$3 \times 5 \times 17$} & $1340 =$ \sol{\dotfill}{$2^2 \times 5 \times 67$} & $720  =$ \sol{\dotfill}{$2^4 \times 3^2 \times 5$} \\
$450  =$ \sol{\dotfill}{$2 \times 3^2 \times 5^2$} & $140  =$ \sol{\dotfill}{$2^2 \times 5 \times 7$} & $378  =$ \sol{\dotfill}{$2 \times 3^3 \times 7$} \\
$84   =$ \sol{\dotfill}{$2^2 \times 3 \times 7$} & $342  =$ \sol{\dotfill}{$2 \times 3^2 \times 19$} & $270  =$ \sol{\dotfill}{$2 \times 3^3 \times 5$} \\
$42   =$ \sol{\dotfill}{$2 \times 3 \times 7$} & $280  =$ \sol{\dotfill}{$2^3 \times 5 \times 7$} & $210  =$ \sol{\dotfill}{$2 \times 3 \times 5 \times 7$} \\
$600  =$ \sol{\dotfill}{$2^3 \times 3 \times 5^2$} & $144  =$ \sol{\dotfill}{$2^4 \times 3^2$} & $117  =$ \sol{\dotfill}{$3^2 \times 13$} \\
$52   =$ \sol{\dotfill}{$52   = 2^2 \times 13$} & $1450 =$ \sol{\dotfill}{$2 \times 5^2 \times 29$} & $480  =$ \sol{\dotfill}{$2^5 \times 3 \times 5$} \\
\end{tabularx}

\end{exercice}
